# Function to display image and its predicted status
def display_image_with_prediction(filepath, model, labels):
    img_array = cv2.imread(filepath) # Read the image
    plt.imshow(cv2.cvtColor(img_array, cv2.COLOR_BGR2RGB)) # Display the image (convert BGR to RGB for matplotlib)
    plt.axis('off') # Hide axes
    plt.show() # Show the image

    # Prepare the image for prediction
    prepared_image = prepare(filepath)

    # Make the prediction
    prediction = model.predict([prepared_image])
    predicted_class_index = np.argmax(prediction)
    predicted_status = labels[predicted_class_index]

    print(f"Predicted status: {predicted_status}")
